        function init() {
            // Role Selection Logic
            roleBtns.forEach(btn => {
                btn.addEventListener('click', () => {
                    roleBtns.forEach(b => {
                        b.classList.remove('border-teal-400', 'bg-teal-50');
                        b.classList.add('border-transparent', 'bg-white');
                    });
                    btn.classList.remove('border-transparent', 'bg-white');
                    btn.classList.add('border-teal-400', 'bg-teal-50');
                    role = btn.dataset.role;
                    
                    btnStart.disabled = false;
                    btnStart.classList.remove('opacity-50', 'cursor-not-allowed');
                    btnStart.classList.add('hover:scale-105', 'active:scale-95');
                });
            });

            // Start Button
            btnStart.addEventListener('click', () => {
                if(role) {
                    screens.start.classList.add('hidden');
                    screens.map.classList.remove('hidden');
                    displayRole.innerText = role;
                    document.getElementById('cert-role').innerText = role;
                    updateTopBar();
                    toggleBGM(true); // Attempt to auto-play on interaction
                }
            });

            // BGM Toggle
            btnBgm.addEventListener('click', () => toggleBGM(!isBgmPlaying));

            // Island Clicks
            islands.forEach(island => {
                island.addEventListener('click', () => {
                    const id = parseInt(island.dataset.id);
                    if (!clearedIslands[id]) {
                        openQuiz(id);
                    }
                });
            });

            // Close Quiz Modal
            document.querySelectorAll('.btn-close-modal').forEach(btn => {
                btn.addEventListener('click', closeQuiz);
            });

            // Chart Toggle
            btnShowChart.addEventListener('click', () => {
                chartContent.classList.toggle('hidden');
            });

            // Feedback OK Click
            btnFeedbackOk.addEventListener('click', closeFeedback);

            // Restart
            document.getElementById('btn-restart').addEventListener('click', () => {
                location.reload();
            });
        }

        // Logic Functions
        function toggleBGM(forcePlay) {
            if (forcePlay) {
                bgm.play().then(() => {
                    isBgmPlaying = true;
                    btnBgm.innerText = '🔊';
                    btnBgm.classList.add('bg-teal-100', 'text-teal-700');
                }).catch(e => {
                    console.log("Audio autoplay blocked", e);
                    isBgmPlaying = false;
                    btnBgm.innerText = '🔇';
                });
            } else {
                bgm.pause();
                isBgmPlaying = false;
                btnBgm.innerText = '🔇';
                btnBgm.classList.remove('bg-teal-100', 'text-teal-700');
                btnBgm.classList.add('bg-slate-100');
            }
        }

        function updateTopBar() {
            // Update Hearts
            let heartsHtml = '';
            for(let i=0; i<3; i++) {
                heartsHtml += i < hp ? '<span class="text-rose-500 drop-shadow-sm">❤️</span>' : '<span class="text-slate-200 grayscale">💔</span>';
            }
            displayHp.innerHTML = heartsHtml;

            // Update Progress
            const pct = (progress / 5) * 100;
            progressBarFill.style.width = `${pct}%`;
            progressText.innerText = `${pct}%`;
        }

        function updateMap() {
            islands.forEach((island, index) => {
                if (clearedIslands[index]) {
                    const pentagon = island.querySelector('.pentagon');
                    const textBadge = island.querySelector('.island-title');
                    
                    pentagon.classList.remove('island-amber', 'animate-bounce-slow');
                    pentagon.classList.add('island-green');
                    pentagon.innerHTML = '<div class="text-4xl text-white drop-shadow-md">✔️</div>';
                    
                    textBadge.classList.remove('text-slate-600', 'border-slate-200');
                    textBadge.classList.add('text-emerald-700', 'bg-emerald-50', 'border-emerald-200');
                    
                    island.classList.remove('cursor-pointer');
                    island.classList.add('cursor-default');
                }
            });
        }

        function openQuiz(index) {
            currentQuestionIndex = index;
            const q = questions[index];
            
            quizTitle.innerText = q.title;
            quizScenario.innerText = q.scenario;
            
            // Handle Chart button
            if (q.hasChart) {
                btnShowChart.classList.remove('hidden');
                btnShowChart.classList.add('flex');
                chartContent.classList.add('hidden');
            } else {
                btnShowChart.classList.add('hidden');
                btnShowChart.classList.remove('flex');
                chartContent.classList.add('hidden');
            }

            // Generate Options
            quizOptions.innerHTML = '';
            q.options.forEach((opt, i) => {
                const btn = document.createElement('button');
                btn.className = 'w-full text-left p-4 rounded-xl border-2 border-slate-100 hover:border-sky-300 hover:bg-sky-50 transition-all group flex items-start gap-4';
                btn.innerHTML = `
                    <span class="shrink-0 bg-slate-100 group-hover:bg-sky-200 text-slate-500 group-hover:text-sky-800 w-8 h-8 flex items-center justify-center rounded-full font-bold">
                        ${i === 0 ? 'A' : 'B'}
                    </span>
                    <span class="text-slate-700 font-medium pt-1">
                        ${opt.text}
                    </span>
                `;
                btn.onclick = () => submitAnswer(opt.isCorrect, opt.feedback);
                quizOptions.appendChild(btn);
            });

            modalQuiz.classList.remove('hidden');
            quizBox.classList.remove('shake');
        }

        function closeQuiz() {
            modalQuiz.classList.add('hidden');
            currentQuestionIndex = -1;
        }

        function submitAnswer(isCorrect, feedback) {
            if (isCorrect) {
                showFeedback(true, "✅ 處置正確！", feedback);
            } else {
                // Apply Shake animation
                quizBox.classList.remove('shake');
                void quizBox.offsetWidth; // trigger reflow
                quizBox.classList.add('shake');
                
                hp--;
                updateTopBar();
                
                if (hp <= 0) {
                    setTimeout(() => {
                        modalQuiz.classList.add('hidden');
                        modalGameover.classList.remove('hidden');
                    }, 500);
                } else {
                    showFeedback(false, "❌ 警告！", feedback);
                }
            }
        }

        function showFeedback(isSuccess, title, msg) {
            modalFeedback.classList.remove('hidden');
            
            // Remove previous styling classes
            feedbackBox.className = "bg-white rounded-3xl p-6 sm:p-8 max-w-sm w-full text-center shadow-2xl pointer-events-auto scale-100 transition-transform duration-300";
            btnFeedbackOk.className = "w-full py-3 rounded-xl font-bold text-white transition-transform active:scale-95 shadow-lg";
            
            feedbackTitle.innerText = title;
            feedbackMsg.innerText = msg;

            if (isSuccess) {
                feedbackIcon.innerText = "🎉";
                feedbackTitle.classList.add('text-emerald-600');
                feedbackTitle.classList.remove('text-rose-600');
                btnFeedbackOk.innerText = "繼續探索";
                btnFeedbackOk.classList.add('bg-emerald-500', 'hover:bg-emerald-600');
                btnFeedbackOk.dataset.action = 'success';
            } else {
                feedbackIcon.innerText = "🚨";
                feedbackTitle.classList.add('text-rose-600');
                feedbackTitle.classList.remove('text-emerald-600');
                btnFeedbackOk.innerText = "了解，重新選擇";
                btnFeedbackOk.classList.add('bg-rose-500', 'hover:bg-rose-600');
                btnFeedbackOk.dataset.action = 'fail';
            }
        }
